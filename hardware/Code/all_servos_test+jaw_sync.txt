#include <Arduino.h>
#include <Wire.h>
#include <Adafruit_PWMServoDriver.h>

Adafruit_PWMServoDriver driverSO = Adafruit_PWMServoDriver(0x40); 
Adafruit_PWMServoDriver driverR  = Adafruit_PWMServoDriver(0x41); // soldered LSB jumper on pwm driver

#define SERVOMID 300   //check position for each servo
const int tilt = 15;

struct SerwoIndywidualne {
  uint8_t pin;           
  unsigned long interwal; 
  unsigned long lastMovement; 
  bool stan;             
  uint8_t idSterownika; 
};

// 19 servos - withouth jaw ones
SerwoIndywidualne myServos[] = {
  {10, 1200, 0, false, 0},
   {9, 1200, 0, false, 0},
    {8, 1500, 0, false, 0},
  {7, 2000, 0, false, 0},
   {6, 500, 0, false, 0},
    {5, 800, 0, false, 0},
  {4, 1200, 0, false, 0},
   {3, 1500, 0, false, 0},
    {2, 2000, 0, false, 0},
  {9, 1200, 0, false, 1},
   {8, 1500, 0, false, 1},
    {7, 2000, 0, false, 1},
  {6, 500, 0, false, 1},
   {5, 800, 0, false, 1},
    {4, 1200, 0, false, 1},
  {3, 1500, 0, false, 1},
   {2, 2000, 0, false, 1},
    {1, 3000, 0, false, 1},
  {0, 3000, 0, false, 1}
};

const int iloscSerw = 19;


unsigned long jawTime = 0;
unsigned long interwalSzczeki = 3000;
bool jawState = false;
const uint8_t jawServoPin1 = 0; // driver SO
const uint8_t jawServoPin2 = 1; // driver SO

void setup() {
  Serial.begin(9600);
  driverSO.begin();
  driverSO.setOscillatorFrequency(27000000);
  driverSO.setPWMFreq(50);

  driverR.begin();
  driverR.setOscillatorFrequency(27000000);
  driverR.setPWMFreq(50);

  delay(10);
}

void loop() {
  unsigned long currentTimeMillis = millis();

  
  if (currentTimeMillis - jawTime >= interwalSzczeki) {
    jawTime = currentTimeMillis;
    
    if (jawState == false) {
   
      driverSO.setPWM(jawServoPin1, 0, SERVOMID + tilt);
      driverSO.setPWM(jawServoPin2, 0, SERVOMID - tilt);
      jawState = true;
    } else {
     
      driverSO.setPWM(jawServoPin1, 0, SERVOMID - tilt);
      driverSO.setPWM(jawServoPin2, 0, SERVOMID + tilt);
      jawState = false;
    }
    Serial.println("Jaw movement - mirroring");
  }


  for (int i = 0; i < iloscSerw; i++) {
    if (currentTimeMillis - myServos[i].lastMovement >= myServos[i].interwal) {
      myServos[i].lastMovement = currentTimeMillis;

      int position = myServos[i].stan ? (SERVOMID - tilt) : (SERVOMID + tilt);
      myServos[i].stan = !myServos[i].stan;

      if (myServos[i].idSterownika == 0) {
        driverSO.setPWM(myServos[i].pin, 0, position);
      } else {
        driverR.setPWM(myServos[i].pin, 0, position);
      }
    }
  }
}
