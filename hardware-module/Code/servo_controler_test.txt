#include <Arduino.h>
#include <Wire.h>
#include <Adafruit_PWMServoDriver.h>

Adafruit_PWMServoDriver pwm = Adafruit_PWMServoDriver();
#define SERVOMIN  100   //  0° 
#define SERVOMAX  480   // 180°

uint8_t servoNumber = 5;

unsigned long currentTime = 0;
unsigned long previousTime = 0;
unsigned long interval = 0;
int stage = 0;

void setAngle(uint8_t channelNumber, uint8_t angle) {
  angle = constrain(angle, 0, 180); 
  uint16_t pulse = map(angle, 0, 180, SERVOMIN, SERVOMAX);
  pwm.setPWM(channelNumber, 0, pulse);
}

void setup() {
  Serial.begin(9600);
  Serial.println("Starting PCA9685 test...");

  pwm.begin();
  pwm.setOscillatorFrequency(27000000);
  pwm.setPWMFreq(50);
  
  // This small setup delay (10ms) is recommended by the library creators
  // to allow the PCA9685 to boot up correctly.
  delay(10); 
}

void loop() {
  
  currentTime = millis();

  if (currentTime - previousTime >= interval) {
    
    previousTime = currentTime; 

    switch (stage) {
      case 0:
        Serial.println("Position: 0 degrees");
        setAngle(servoNumber, 0);
        interval = 1500; 
        stage = 1;       
        break;

      case 1:
        Serial.println("Position: 180 degrees");
        setAngle(servoNumber, 180);
        interval = 1500; 
        stage = 2;       
        break;

      case 2:
        Serial.println("Position: 90 degrees (center). Now 5 seconds pause...");
        setAngle(servoNumber, 90);
        interval = 5000; 
        stage = 0;        
        break;
    }
  }
}
